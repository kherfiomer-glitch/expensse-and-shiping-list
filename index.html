const SHEET_TRANSACTIONS = "Transactions"; const SHEET_PRODUCTS = "Products"; const SHEET_SHOPPING = "Shopping"; /* ========================= SPREADSHEET ========================= */ function getSpreadsheet() { return SpreadsheetApp.getActiveSpreadsheet(); } function getOrCreateSheet(name, headers) { const ss = getSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.setFrozenRows(1); } return sheet; } /* ========================= SETUP ========================= */ function setupSheets() { getOrCreateSheet( SHEET_TRANSACTIONS, ["Date", "Description", "In", "Out", "Added By"] ); getOrCreateSheet( SHEET_PRODUCTS, ["Product", "Category", "Unit", "Price DA"] ); getOrCreateSheet( SHEET_SHOPPING, ["Date Needed", "Product", "Quantity", "Unit", "Unit Price", "Total", "Added By"] ); } /* ========================= GET DATA ========================= */ function doGet(e) { try { setupSheets(); const data = { success: true, transactions: getTransactions(), products: getProducts(), shopping: getShopping() }; /* JSONP support. This allows the website to read the Google Apps Script data even when the browser blocks normal fetch/CORS requests. */ if ( e && e.parameter && e.parameter.callback ) { const callback = e.parameter.callback; return ContentService .createTextOutput( callback + "(" + JSON.stringify(data) + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } return jsonResponse(data); } catch (error) { const data = { success: false, error: error.message }; if ( e && e.parameter && e.parameter.callback ) { const callback = e.parameter.callback; return ContentService .createTextOutput( callback + "(" + JSON.stringify(data) + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } return jsonResponse(data); } } /* ========================= POST ========================= */ function doPost(e) { try { const data = JSON.parse( (e.postData && e.postData.contents) || "{}" ); setupSheets(); if (data.action === "addTransaction") { return jsonResponse( addTransaction(data) ); } if (data.action === "addShopping") { return jsonResponse( addShopping(data) ); } if (data.action === "deleteShopping") { return jsonResponse( deleteShopping(data) ); } throw new Error("Unknown action."); } catch (error) { return jsonResponse({ success: false, error: error.message }); } } /* ========================= TRANSACTIONS ========================= */ function addTransaction(data) { if ( !data.amount || !data.description || !data.date || !data.person ) { throw new Error( "Please fill all transaction fields." ); } if ( data.type !== "in" && data.type !== "out" ) { throw new Error( "Invalid transaction type." ); } const sheet = getOrCreateSheet( SHEET_TRANSACTIONS, ["Date", "Description", "In", "Out", "Added By"] ); const amount = Number(data.amount); if (amount <= 0) { throw new Error( "Amount must be greater than 0." ); } sheet.appendRow([ new Date( data.date + "T12:00:00" ), String(data.description), data.type === "in" ? amount : "", data.type === "out" ? amount : "", String(data.person) ]); return { success: true }; } /* ========================= SHOPPING ========================= */ function addShopping(data) { if ( !data.date || !data.product || !data.quantity || !data.person ) { throw new Error( "Please fill all shopping fields." ); } const quantity = Number( data.quantity ); const unitPrice = Number( data.unitPrice ); if (quantity <= 0) { throw new Error( "Quantity must be greater than 0." ); } if ( unitPrice < 0 || isNaN(unitPrice) ) { throw new Error( "Invalid unit price." ); } const total = quantity * unitPrice; const sheet = getOrCreateSheet( SHEET_SHOPPING, [ "Date Needed", "Product", "Quantity", "Unit", "Unit Price", "Total", "Added By" ] ); sheet.appendRow([ new Date( data.date + "T12:00:00" ), String(data.product), quantity, String(data.unit || ""), unitPrice, total, String(data.person) ]); return { success: true }; } /* ========================= DELETE SHOPPING ========================= */ function deleteShopping(data) { const rowNumber = Number( data.rowNumber ); if ( !rowNumber || rowNumber < 2 ) { throw new Error( "Invalid shopping row." ); } const sheet = getOrCreateSheet( SHEET_SHOPPING, [ "Date Needed", "Product", "Quantity", "Unit", "Unit Price", "Total", "Added By" ] ); if ( rowNumber <= sheet.getLastRow() ) { sheet.deleteRow( rowNumber ); } return { success: true }; } /* ========================= READ TRANSACTIONS ========================= */ function getTransactions() { const sheet = getOrCreateSheet( SHEET_TRANSACTIONS, [ "Date", "Description", "In", "Out", "Added By" ] ); const values = sheet.getDataRange().getValues(); if (values.length <= 1) { return []; } return values .slice(1) .map(function(row, index) { const moneyIn = Number(row[2]) || 0; const moneyOut = Number(row[3]) || 0; return { rowNumber: index + 2, id: index + 1, date: formatDate( row[0] ), description: String(row[1] || ""), type: moneyIn > 0 ? "in" : "out", amount: moneyIn > 0 ? moneyIn : moneyOut, person: String(row[4] || "") }; }) .filter(function(x) { return ( x.date && x.description ); }); } /* ========================= READ PRODUCTS ========================= */ function getProducts() { const sheet = getOrCreateSheet( SHEET_PRODUCTS, [ "Product", "Category", "Unit", "Price DA" ] ); const lastRow = sheet.getLastRow(); if (lastRow <= 1) { return []; } /* IMPORTANT: This reads the CURRENT values directly from Google Sheets. So if you change: أرز | مواد غذائية | كغ | 350 to: أرز | مواد غذائية | كغ | 400 the website will receive 400. */ const values = sheet .getRange( 2, 1, lastRow - 1, 4 ) .getValues(); return values .map(function(row, index) { return { rowNumber: index + 2, product: String( row[0] || "" ).trim(), category: String( row[1] || "" ).trim(), unit: String( row[2] || "" ).trim(), price: Number( row[3] ) || 0 }; }) .filter(function(x) { return x.product; }); } /* ========================= READ SHOPPING ========================= */ function getShopping() { const sheet = getOrCreateSheet( SHEET_SHOPPING, [ "Date Needed", "Product", "Quantity", "Unit", "Unit Price", "Total", "Added By" ] ); const values = sheet.getDataRange().getValues(); if (values.length <= 1) { return []; } return values .slice(1) .map(function(row, index) { return { rowNumber: index + 2, date: formatDate( row[0] ), product: String( row[1] || "" ), quantity: Number( row[2] ) || 0, unit: String( row[3] || "" ), unitPrice: Number( row[4] ) || 0, total: Number( row[5] ) || 0, person: String( row[6] || "" ) }; }) .filter(function(x) { return x.product; }); } /* ========================= DATE ========================= */ function formatDate(value) { if ( value instanceof Date && !isNaN(value.getTime()) ) { return Utilities.formatDate( value, Session.getScriptTimeZone(), "yyyy-MM-dd" ); } return String( value || "" ); } /* ========================= JSON RESPONSE ========================= */ function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); }
